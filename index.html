// Bộ nhớ đệm Client: Khi người dùng đổi tab qua lại, phản hồi ngay lập tức 0ms const memoryCache = new Map(); const CACHE_TTL = 3 * 60 * 1000; // 3 phút google.script.run = new Proxy(google.script.run, { get(target, prop) { if (prop in target) return target[prop]; return function (...args) { const onSuccess = target._success || (() => {}); const onFailure = target._failure || console.error; target._success = null; target._failure = null; const readOnlyActions = ['getClasses', 'getStudents', 'getDashboardData', 'getTeacherDashboardData', 'getMonthlyTuition', 'getTeachers']; const isReadOnly = readOnlyActions.includes(prop); const cacheKey = prop + JSON.stringify(args); // Nếu đã có cache còn hạn -> Trả về ngay lập tức if (isReadOnly && memoryCache.has(cacheKey)) { const cached = memoryCache.get(cacheKey); if (Date.now() - cached.time < CACHE_TTL) { onSuccess(cached.data); return; } } fetch(APPS_SCRIPT_URL, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ action: prop, args: args }) }) .then(res => res.json()) .then(data => { if (isReadOnly && data && data.success) { memoryCache.set(cacheKey, { time: Date.now(), data: data }); } if (!isReadOnly) { memoryCache.clear(); // Xóa sạch bộ nhớ tạm khi vừa thêm/sửa/xóa } onSuccess(data); }) .catch(err => onFailure(err)); }; } });
